----- src/features/dashboard/hooks/useTransactionsBulk.ts -----
import { keepPreviousData, useQuery } from "@tanstack/react-query";
import { getTransactions } from "../api/dashboardApi";

// Large enough to cover realistic personal transaction volume for a date
// range in one request, small enough to stay a bounded, predictable payload
// rather than truly unbounded ("fetch everything, however much that is").
const BULK_FETCH_LIMIT = 500;

/**
 * Fetches every transaction in a date range in one bulk page, so the charts
 * can be filtered/aggregated client-side (search, category, status, amount)
 * without re-hitting the network on every keystroke — only the date range
 * (driven by the period selector) changes the query key here. This is
 * intentionally separate from useTransactions, which stays page-by-page for
 * the table itself.
 */
export function useTransactionsBulk(params: {
  start_date?: string;
  end_date?: string;
}) {
  return useQuery({
    queryKey: ["dashboard", "transactions-bulk", params],
    queryFn: () =>
      getTransactions({
        ...params,
        page: 1,
        limit: BULK_FETCH_LIMIT,
        sort_by: "timestamp",
        sort_order: "asc",
      }),
    // Keep the previous range's data visible while a new date range loads,
    // instead of the charts flashing empty.
    placeholderData: keepPreviousData,
    staleTime: 60_000,
  });
}

export type TransactionStatus = "SUCCESS" | "FAILED" | "PENDING";

export type PaymentMethod =
  | "Credit Card"
  | "Debit Card"
  | "UPI"
  | "Netbanking"
  | "Wallet";

export type Transaction = {
  id: string;
  timestamp: string;
  merchant: string;
  category: string;
  amount: number;
  currency: string;
  status: TransactionStatus;
  payment_method: PaymentMethod;
};

/**
 * Matches the documented /api/v1/transactions response exactly:
 * { items, total, page, page_count }. There is no `limit` echoed back and
 * no `total_pages` — callers that need the page size already know it (they
 * sent it), and `page_count` is the doc's name for total pages.
 */
export type TransactionsResponse = {
  items: Transaction[];
  total: number;
  page: number;
  page_count: number;
};

export type TransactionQueryParams = {
  page?: number;
  limit?: number;
  category?: string;
  status?: string;
  payment_method?: string;
  search?: string;
  start_date?: string;
  end_date?: string;
  sort_by?: string;
  sort_order?: "asc" | "desc";
};

/**
 * The documented /analytics/spend endpoint only accepts a date range — no
 * search, category, status, or amount filters. Those filters stay
 * table-only; the trend/category charts are date-scoped and, for anything
 * finer, derived client-side from the bulk transaction set (see
 * useFilteredSummary in features/dashboard/utils).
 */
export type SpendAnalyticsQueryParams = {
  start_date?: string;
  end_date?: string;
};

export type CategoryBreakdownEntry = {
  category: string;
  total_amount: number;
  transaction_count?: number;
};

export type MonthlyTrendEntry = {
  month: string;
  total_amount: number;
  transaction_count?: number;
};

export type TopMerchantEntry = {
  merchant: string;
  total_amount: number;
  transaction_count?: number;
};

/**
 * Matches the documented /api/v1/analytics/spend response:
 * { total_spend, category_breakdown, monthly_trends, top_merchants }.
 * The previous total_transactions/successful_transactions/coins_earned
 * fields aren't part of this response — those metric cards now derive
 * from the bulk transaction set instead (see useFilteredSummary in
 * features/dashboard/utils).
 */
export type SpendAnalyticsResponse = {
  total_spend: number;
  category_breakdown: CategoryBreakdownEntry[];
  monthly_trends: MonthlyTrendEntry[];
  top_merchants: TopMerchantEntry[];
};

import { useEffect, useState } from "react";
import {
  fetchRewardBalance,
  fetchRewardCatalog,
  redeemReward,
} from "../api";
import type { RewardItem } from "../types";
import { FilterTabs } from "../components/FilterTabs";
import { RewardCard } from "../components/RewardCard";

function StatsCards({
  balance,
  isLoading,
  isError,
}: {
  balance: number | null;
  isLoading: boolean;
  isError: boolean;
}) {
  const safeBalance = Number(balance) || 0;

  return (
    <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
      <div className="rounded-xl border border-gray-800 bg-[#111] p-5">
        <p className="text-sm text-gray-400">Coin balance</p>

        <p className="mt-2 text-2xl font-bold">
          {isLoading
            ? "—"
            : isError
              ? "Unavailable"
              : safeBalance.toLocaleString("en-IN")}
        </p>
      </div>
    </div>

export function RewardsPage() {
  const [balance, setBalance] = useState<number | null>(null);
  const [catalog, setCatalog] = useState<RewardItem[]>([]);
  const [isLoadingBalance, setIsLoadingBalance] = useState(true);
  const [isLoadingCatalog, setIsLoadingCatalog] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [redeemingId, setRedeemingId] = useState<number | null>(null);

  useEffect(() => {
    const loadData = async () => {
      try {
        const [balanceData, catalogData] = await Promise.all([
          fetchRewardBalance(),
          fetchRewardCatalog(),
        ]);

        setBalance(Number(balanceData.coin_balance) || 0);

        setCatalog(catalogData.rewards);
      } catch (err) {
        setError(
          err instanceof Error
            ? err.message
            : "Failed to load rewards data",
        );
      } finally {
        setIsLoadingBalance(false);
        setIsLoadingCatalog(false);
      }
    };

    loadData();
  }, []);

  const handleRedeem = async (reward: RewardItem) => {
    if (balance === null) {
      return;
    }

    const coinCost = reward.cost_coins;
    const previousBalance = balance;

    if (coinCost > balance) {
      setError("You do not have enough coins to redeem this reward.");
      return;
    }

    setBalance((previous) => (previous ?? 0) - coinCost);
    setRedeemingId(reward.id);
    setError(null);

    try {
      await redeemReward(reward.id);

      const [balanceData, catalogData] = await Promise.all([
        fetchRewardBalance(),
        fetchRewardCatalog(),
      ]);

      setBalance(balanceData.coin_balance);
      setCatalog(catalogData.rewards);
    } catch (err) {
      setBalance(previousBalance);

      setError(
        err instanceof Error
          ? err.message
          : "Redemption failed. Balance restored.",
      );
    } finally {
      setRedeemingId(null);
    }
  };

  return (
    <div className="min-h-screen bg-black p-8 text-white">
      <div className="mb-8 mt-6">
        <h1 className="text-3xl font-bold">Rewards</h1>

        <p className="mt-2 text-gray-400">
          Redeem your coins for exciting vouchers, cashback and more.
        </p>
      </div>

      {error && (
        <div className="mb-6 rounded-lg border border-red-500/30 bg-red-500/10 p-4 text-red-400">
          {error}
        </div>
      )}

      <div className="flex flex-col gap-6">
        <StatsCards
          balance={balance}
          isLoading={isLoadingBalance}
          isError={Boolean(error)}
        />

        <FilterTabs />

        {isLoadingCatalog ? (
          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4">
            {[...Array(8)].map((_, index) => (
              <div
                key={index}
                className="h-64 animate-pulse rounded-xl border border-gray-800 bg-[#111]"
              />
            ))}
          </div>
        ) : (
          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4">
            {catalog.map((reward) => (
              <RewardCard
                key={reward.id}
                reward={reward}
                onRedeem={handleRedeem}
                isRedeeming={redeemingId === reward.id}
              />
            ))}
          </div>
        )}
      </div>
    </div>
  );
}

        const [balanceData, catalogData] = await Promise.all([
          fetchRewardBalance(),
          fetchRewardCatalog(),
        ]);
        setBalance(Number(balanceData.coin_balance) || 0);
        const rewardsData = Array.isArray(catalogData) ? catalogData : catalogData?.rewards;
        setCatalog(Array.isArray(rewardsData) ? rewardsData : []);
      } catch (err) {
        setError(err instanceof Error ? err.message : "Failed to load rewards data");
      } finally {
        setIsLoadingBalance(false);
        setIsLoadingCatalog(false);
      }
    };
    loadData();
  }, []);

  const handleRedeem = async (reward: RewardItem) => {
    if (balance === null) return;
    const coinCost = Number(reward.cost_coins) || 0;
    const previousBalance = balance;
    setBalance(prev => (Number(prev) || 0) - coinCost);
    setRedeemingId(reward.id);
    setError(null);
    try {
      await redeemReward(reward.id);
      const [balanceData, catalogData] = await Promise.all([
        fetchRewardBalance(),
        fetchRewardCatalog(),
      ]);
      setBalance(Number(balanceData.coin_balance) || 0);
      const rewardsData = Array.isArray(catalogData) ? catalogData : catalogData?.rewards;
      setCatalog(Array.isArray(rewardsData) ? rewardsData : []);
    } catch (err) {
      setBalance(previousBalance);
      setError(err instanceof Error ? err.message : "Redemption failed. Balance restored.");
    } finally {
      setRedeemingId(null);
    }
  };

  return (
    <div className="p-8 bg-black text-white min-h-screen">
      <div className="mt-6 mb-8">
        <h1 className="text-3xl font-bold">Rewards</h1>
        <p className="text-gray-400 mt-2">Redeem your coins for exciting vouchers, cashback and more.</p>
      </div>

      {error && (
        <div className="mb-6 p-4 bg-red-500/10 border border-red-500/30 text-red-400 rounded-lg">
          {error}
        </div>
      )}

      <div className="grid grid-cols-12 gap-6">
        <div className="col-span-12 lg:col-span-9 flex flex-col gap-6">
          <StatsCards balance={balance} isLoading={isLoadingBalance} isError={!!error} />
          <FilterTabs />
          {isLoadingCatalog ? (
            <div className="grid grid-cols-1 sm:grid-cols-2 xl:grid-cols-4 gap-4">
              {[...Array(8)].map((_, i) => (
                <div key={i} className="h-64 bg-[#111] border border-gray-800 rounded-xl animate-pulse" />
              ))}
            </div>
          ) : (
            <div className="grid grid-cols-1 sm:grid-cols-2 xl:grid-cols-4 gap-4">
              {catalog.map((reward) => (
                <RewardCard key={reward.id} reward={reward} onRedeem={handleRedeem} isRedeeming={redeemingId === reward.id} />
              ))}
            </div>
          )}
        </div>

        <div className="col-span-12 lg:col-span-3">
          <RecentRedemptions />
        </div>
      </div>
    </div>
  );
}

import { useMemo, useState } from "react";
import { useNavigate } from "react-router-dom";
import { useDebouncedValue } from "@/hooks/useDebouncedValue";
import { DashboardHeader } from "@/features/dashboard/components/DashboardHeader";
import { DashboardMetricCard } from "@/features/dashboard/components/DashboardMetricCard";
import { PaymentCategoriesChart } from "@/features/dashboard/components/PaymentCategoriesChart";
import { RewardsPromoCard } from "@/features/dashboard/components/RewardsPromoCard";
import { SpendingChart } from "@/features/dashboard/components/SpendingChart";
import {
  TRANSACTIONS_SORT_OPTIONS,
  TransactionsTable,
} from "@/features/dashboard/components/TransactionsTable";
import {
  TransactionsFilters,
  type TransactionsFiltersValue,
} from "@/features/dashboard/components/TransactionsFilters";
import { useRewardBalance } from "@/features/dashboard/hooks/useRewardBalance";
import { useRewardsCatalog } from "@/features/dashboard/hooks/useRewardsCatalog";
import { useTransactionsBulk } from "@/features/dashboard/hooks/useTransactionsBulk";
import {
  DEFAULT_ANALYTICS_PERIOD,
  getPeriodDateRange,
  getPeriodOption,
} from "@/features/dashboard/utils/period";
import {
  applyClientFilters,
  buildCategoryBreakdownFromTransactions,
  buildTrendFromTransactions,
  getDateRangeForTrendLabel,
} from "@/features/dashboard/utils/trend";
import { useFilteredSummary } from "@/features/dashboard/utils/useFilteredSummary";

const PAGE_SIZE = 10;

const EMPTY_FILTERS: TransactionsFiltersValue = {
  search: "",
  category: "",
  status: "",
  minAmount: "",
  maxAmount: "",
  startDate: "",
  endDate: "",
};

/**
 * The whole dashboard is driven by one bulk transaction fetch per date
 * range (useTransactionsBulk) — search/category/status/amount filters,
 * sorting, and pagination are all applied in-memory from that set. This is
 * what makes the charts, metric cards, and table all react together to
 * search/filter changes and to clicking a chart point/slice: they all read
 * from the same filtered+sorted array, just sliced differently.
 */
export function DashboardPage() {
  const navigate = useNavigate();
  const [filters, setFilters] = useState<TransactionsFiltersValue>(EMPTY_FILTERS);
  const [page, setPage] = useState(1);
  const [sortValue, setSortValue] = useState(TRANSACTIONS_SORT_OPTIONS[0].value);
  const [period, setPeriod] = useState(DEFAULT_ANALYTICS_PERIOD.value);

  const sortOption =
    TRANSACTIONS_SORT_OPTIONS.find((option) => option.value === sortValue) ??
    TRANSACTIONS_SORT_OPTIONS[0];

  const periodOption = getPeriodOption(period);
  // Determine the effective date range for the API call.
// If the user has manually set a start or end date, use those instead of the period selector.
const effectiveDateRange = useMemo(() => {
  if (filters.startDate || filters.endDate) {
    return {
      start_date: filters.startDate || undefined,
      end_date: filters.endDate || undefined,
    };
  }
  return getPeriodDateRange(period);
}, [filters.startDate, filters.endDate, period]);

  // Debounce the search text so the *derived* views don't thrash on every
  // keystroke's intermediate value. This no longer gates a network request
  // (there isn't one per keystroke) — it just smooths the recompute.
  const debouncedSearch = useDebouncedValue(filters.search, 150);

  // One bulk fetch per date range (period selector), cached by React Query.
  // Search/category/status/amount/sort/page never trigger a new request —
  // they're all applied client-side below — which is what keeps typing and
  // chart clicks instant even with a large transaction volume.
  const {
    data: bulkData,
    isLoading: isBulkLoading,
    isFetching: isBulkFetching,
    isError: isBulkError,
  } = useTransactionsBulk(effectiveDateRange);

  const bulkTransactions = useMemo(() => bulkData?.items ?? [], [bulkData]);

  const effectiveFilters = useMemo(
    () => ({ ...filters, search: debouncedSearch }),
    [filters, debouncedSearch],
  );

  // The single source of truth for "what matches the current filters" —
  // charts, metric cards, and the table all read from this one filtered
  // view instead of drifting out of sync with each other.
  const summary = useFilteredSummary(bulkTransactions, effectiveFilters);

  const sortedTransactions = useMemo(() => {
    const sorted = [...summary.filteredTransactions];

    sorted.sort((a, b) => {
      const direction = sortOption.sortOrder === "asc" ? 1 : -1;

      if (sortOption.sortBy === "amount") {
        return (a.amount - b.amount) * direction;
      }

      return (
        (new Date(a.timestamp).getTime() - new Date(b.timestamp).getTime()) *
        direction
      );
    });

    return sorted;
  }, [summary.filteredTransactions, sortOption]);

  const totalTransactions = sortedTransactions.length;
  const totalPages = Math.max(1, Math.ceil(totalTransactions / PAGE_SIZE));
  const safePage = Math.min(page, totalPages);
  const pagedTransactions = useMemo(
    () =>
      sortedTransactions.slice(
        (safePage - 1) * PAGE_SIZE,
        safePage * PAGE_SIZE,
      ),
    [sortedTransactions, safePage],
  );

  const {
    data: rewardBalance,
    isLoading: isRewardBalanceLoading,
    isError: isRewardBalanceError,
  } = useRewardBalance();

  const { data: rewardsCatalog } = useRewardsCatalog();

  const spendingData = useMemo(
    () =>
      buildTrendFromTransactions(
        summary.filteredTransactions,
        periodOption.granularity,
      ),
    [summary.filteredTransactions, periodOption.granularity],
  );

  // Category options for the filter dropdown come from the full bulk set
  // (not the filtered one) so picking a category never removes other
  // categories from the list.
  const categoryOptions = useMemo(
    () =>
      Array.from(new Set(bulkTransactions.map((t) => t.category))).sort(),
    [bulkTransactions],
  );

  const chartFilters = useMemo(
    () => ({ ...effectiveFilters, category: '' }),
    [effectiveFilters]
  );

  const chartFilteredTransactions = useMemo(
    () => applyClientFilters(bulkTransactions, chartFilters),
    [bulkTransactions, chartFilters]
  );

  const chartCategoryBreakdown = useMemo(
    () => buildCategoryBreakdownFromTransactions(chartFilteredTransactions),
    [chartFilteredTransactions]
  );

  function handleFiltersChange(next: TransactionsFiltersValue) {
    setFilters(next);
    setPage(1);
  }

  function handleClearFilters() {
    setFilters(EMPTY_FILTERS);
    setPage(1);
  }

  function handleSortChange(value: string) {
    setSortValue(value);
    setPage(1);
  }

  // Clicking a trend point narrows every derived view (table, metrics,
  // donut) to that month/year's date range.
  function handleTrendPointClick(label: string) {
    const range = getDateRangeForTrendLabel(label);
    if (!range) return;

    setFilters((prev) => ({
      ...prev,
      startDate: range.startDate,
      endDate: range.endDate,
    }));
    setPage(1);
  }

  // Clicking a category slice/legend row narrows the table to that
  // category — clicking the already-selected category clears it again.
  function handleCategoryClick(category: string) {
    setFilters((prev) => ({
      ...prev,
      category: prev.category === category ? "" : category,
    }));
    setPage(1);
  }

  // Only the bulk fetch (which everything derives from) and reward balance
  // gate the whole page. There's no separate transactions request anymore,
  // so there's no second loading/error state to reconcile against this one.
  if (isBulkLoading || isRewardBalanceLoading) {
    return (
      <div className="flex min-h-[400px] items-center justify-center">
        <p className="text-sm text-white/60">Loading dashboard...</p>
      </div>
    );
  }

  if (isBulkError || isRewardBalanceError) {
    return (
      <div className="flex min-h-[400px] items-center justify-center">
        <p className="text-sm text-red-400">Unable to load dashboard data.</p>
      </div>
    );
  }

  return (
    <div className="mx-auto w-full max-w-[1440px] space-y-6">
      <DashboardHeader
        userName="John"
        coinBalance={rewardBalance?.balance ?? 0}
        notificationCount={3}
      />

      <section
        aria-label="Financial overview"
        className="grid grid-cols-1 gap-4 sm:grid-cols-2 xl:grid-cols-4"
      >
        <DashboardMetricCard
          title="Total spent"
          value={`₹${summary.totalSpend.toLocaleString("en-IN", {
            maximumFractionDigits: 0,
          })}`}
          icon="spent"
        />

        <DashboardMetricCard
          title="Successful payments"
          value={summary.successfulTransactions.toLocaleString("en-IN")}
          icon="payments"
        />

        <DashboardMetricCard
          title="Coins earned"
          value={summary.coinsEarned.toLocaleString("en-IN")}
          icon="coins"
        />

        <DashboardMetricCard
          title="Average payment"
          value={
            summary.successfulTransactions > 0
              ? `₹${Math.round(
                  summary.totalSpend / summary.successfulTransactions,
                ).toLocaleString("en-IN")}`
              : "₹0"
          }
          icon="average"
        />
      </section>

      <section
        aria-label="Spending breakdown"
        className="grid grid-cols-1 gap-6 xl:grid-cols-[minmax(0,0.85fr)_minmax(0,1.15fr)]"
      >
        <PaymentCategoriesChart
          categories={chartCategoryBreakdown}        
          period={period}
          onPeriodChange={setPeriod}
          isFetching={isBulkFetching}
          onCategoryClick={handleCategoryClick}
          selectedCategory={filters.category}     
        />
        <SpendingChart
          data={spendingData}
          period={period}
          onPeriodChange={setPeriod}
          isFetching={isBulkFetching}
          onPointClick={handleTrendPointClick}
        />
      </section>

      <TransactionsFilters
        value={filters}
        categories={categoryOptions}
        onChange={handleFiltersChange}
        onClearAll={handleClearFilters}
      />

      <section
        aria-label="Transactions and rewards"
        className="grid grid-cols-1 gap-6 xl:grid-cols-[minmax(0,1fr)_320px]"
      >
        <TransactionsTable
          transactions={pagedTransactions}
          total={totalTransactions}
          page={safePage}
          limit={PAGE_SIZE}
          totalPages={totalPages}
          sortValue={sortValue}
          onSortChange={handleSortChange}
          onPageChange={setPage}
          isLoading={isBulkLoading}
          isFetching={isBulkFetching}
          isError={isBulkError}
        />

        <RewardsPromoCard
          rewards={rewardsCatalog?.rewards ?? []}
          onExploreRewards={() => navigate("/rewards")}
        />
      </section>
    </div>
  );
}
